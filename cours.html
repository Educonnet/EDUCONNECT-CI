javascript async function terminerLecon(courseId) { try { const supabase = window.eduSupabase; if (!supabase) { alert("La connexion à Supabase n'est pas disponible."); return; } const { data, error: userError } = await supabase.auth.getUser(); if (userError || !data?.user) { alert("Connecte-toi pour enregistrer ta progression."); window.location.href = "connexion.html"; return; } // Vérification spéciale pour Initiation à l'informatique if (courseId === "informatique") { const { data: progressionQuiz, error: erreurQuiz } = await supabase .from("course_progress") .select("score") .eq("user_id", data.user.id) .eq("course_id", "informatique") .maybeSingle(); if (erreurQuiz) { console.error("Erreur de lecture du score :", erreurQuiz); alert("Impossible de vérifier ton score. Réessaie."); return; } if (!progressionQuiz || progressionQuiz.score !== 3) { alert( "Tu dois obtenir 3/3 au quiz avant de terminer cette formation." ); return; } } const { data: resultat, error } = await supabase.rpc( "terminer_formation_informatique" ); if (error || resultat !== true) { console.error("Erreur de sauvegarde :", error); alert( error?.message || "La formation n'a pas pu être validée. Vérifie que tu as obtenu 3/3 au quiz." ); return; } alert("Bravo ! Ta formation a été validée et ta progression enregistrée."); } catch (error) { console.error("Erreur inattendue :", error); alert("Une erreur est survenue. Vérifie ta connexion puis réessaie."); }
