javascript async function terminerLecon(courseId) { try { const supabase = window.eduSupabase; if (!supabase) { alert("La connexion à Supabase n'est pas disponible."); return; } const { data, error: userError } = await supabase.auth.getUser(); if (userError || !data?.user) { alert("Connecte-toi pour enregistrer ta progression."); window.location.href = "connexion.html"; return; } // Vérification spéciale pour Initiation à l'informatique if (courseId === "informatique") { const { data: progressionQuiz, error: erreurQuiz } = await supabase .from("course_progress") .select("score") .eq("user_id", data.user.id) .eq("course_id", "informatique") .maybeSingle(); if (erreurQuiz) { console.error("Erreur de lecture du score :", erreurQuiz); alert("Impossible de vérifier ton score. Réessaie."); return; } if (!progressionQuiz || progressionQuiz.score !== 3) { alert( "Tu dois obtenir 3/3 au quiz avant de terminer cette formation." ); return; } } const { error } = await supabase .from("course_progress") .upsert( { user_id: data.user.id, course_id: courseId, progress: 100, completed: true, updated_at: new Date().toISOString() }, { onConflict: "user_id,course_id" } ); if (error) { console.error("Erreur de sauvegarde :", error); alert("La sauvegarde a échoué. Vérifie ta connexion et réessaie."); return; } alert("Bravo ! Ta progression a été enregistrée."); } catch (error) { console.error("Erreur inattendue :", error); alert("Une erreur est survenue. Vérifie ta connexion puis réessaie."); }
