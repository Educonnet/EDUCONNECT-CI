const formulaire = document.getElementById("formReset"); const message = document.getElementById("message"); const bouton = document.getElementById("boutonReset"); const supabase = window.eduSupabase; // Vérification du lien de récupération async function initialiser() { try { if (!supabase) { throw new Error("Service Supabase indisponible."); } const url = new URL(window.location.href); // Vérifier les éventuelles erreurs renvoyées par Supabase const hashParams = new URLSearchParams( url.hash.substring(1) ); const errorCode = url.searchParams.get("error_code") || hashParams.get("error_code"); const errorDescription = url.searchParams.get("error_description") || hashParams.get("error_description"); if (errorCode) { throw new Error( errorDescription || "Le lien de récupération est invalide ou expiré." ); } // Laisser Supabase traiter automatiquement le retour // du lien de récupération (PKCE ou jetons). const { data, error } = await supabase.auth.getSession(); if (error) throw error; if (!data.session) { throw new Error( "Aucune session de récupération détectée. Demande un nouveau lien." ); } formulaire.hidden = false; message.textContent = "Lien vérifié. Tu peux maintenant choisir ton nouveau mot de passe."; } catch (erreur) { console.error("Erreur de récupération :", erreur); message.textContent = erreur.message; formulaire.hidden = true; } } // Enregistrement du nouveau mot de passe formulaire.addEventListener("submit", async function (event) { event.preventDefault(); const nouveau = document.getElementById( "nouveauMotDePasse" ).value; const confirmation = document.getElementById( "confirmationMotDePasse" ).value; if (nouveau.length < 8) { message.textContent = "Le mot de passe doit contenir au moins 8 caractères."; return; } if (nouveau !== confirmation) { message.textContent = "Les deux mots de passe ne correspondent pas."; return; } try { bouton.disabled = true; bouton.textContent = "Enregistrement..."; message.textContent = "Mise à jour du mot de passe en cours..."; const { error } = await supabase.auth.updateUser({ password: nouveau }); if (error) throw error; message.textContent = "Mot de passe modifié avec succès. Redirection vers la connexion..."; // Fermer la session de récupération await supabase.auth.signOut(); window.location.href = "connexion.html"; } catch (erreur) { console.error("Erreur de modification :", erreur); message.textContent = "Impossible de modifier le mot de passe : " + erreur.message; bouton.disabled = false; bouton.textContent = "Enregistrer le nouveau mot de passe"; } }); initialiser();
